# Run this setup in each terminal before a command below.

# Fresh-run destinations must not contain an earlier run.

# Reconstructed stage 1 command matching saved settings; separate output.
"$PYTHON" train.py \
  --stage 1 \
  --annotations "$TRAIN" --frames-root "$FRAMES" \
  --val-annotations "$VAL" --val-frames-root "$FRAMES" \
  --max-val-videos 100 --subset-seed 7 \
  --seed 0 --epochs 50 --batch-size 32 --num-workers 8 \
  --prefetch-factor 1 --sharing-strategy file_system \
  --cudnn-benchmark --learning-rate 0.003 --device cuda \
  --backbone-weights weights/mobilenet_v1_1.0_224_backbone.pt \
  --checkpoint-out runs/big_stage1_reproduction/stage1.pt

# Big model stage 2 initialization

# Start a fresh stage 2 from the existing best stage 1 checkpoint.
"$PYTHON" train.py \
  --stage 2 \
  --annotations "$TRAIN" --frames-root "$FRAMES" \
  --val-annotations "$VAL" --val-frames-root "$FRAMES" \
  --val-frames-per-video 20 --subset-seed 7 \
  --seed 0 --epochs 50 --batch-size 16 --clip-stride 10 \
  --num-workers 4 --prefetch-factor 1 \
  --sharing-strategy file_system --cudnn-benchmark \
  --frozen-backbone-batch-size 0 \
  --learning-rate 0.003 --device cuda \
  --checkpoint-in "$ORIGINAL/runs/stage1_best.pt" \
  --checkpoint-out runs/big_stage2_reproduction/stage2.pt

# Big model stage 2 continuation

# Continue the existing big stage 2 run; do not start a duplicate job.

  --stage 2 \

  --val-annotations "$VAL" --val-frames-root "$FRAMES" \

  --resume runs/big_stage2_continued/stage2.pt \
  --checkpoint-out runs/big_stage2_continued/stage2.pt

# Small model training

# Original small-model configuration: 10 zero-memory + 5 clean-memory.
# The saved run reached epoch 6 before the joint transition.

  --fast-init slice-big \
  --annotations "$TRAIN" --frames-root "$FRAMES" \
  --val-annotations "$VAL" --val-frames-root "$FRAMES" \
  --val-frames-per-video 20 --subset-seed 7 --seed 0 \
  --batch-size 16 --clip-length 10 --clip-stride 10 \
  --num-workers 2 --prefetch-factor 1 \
  --sharing-strategy file_system \
  --reset-fast-bn --fast-bn-momentum 0.01 \
  --zero-memory-epochs 10 --clean-memory-epochs 5 \
  --warmup-interval 4 \
  --learning-rate 0.0001 --fast-learning-rate 0.001 \
  --schedules fixed4 big small \
  --memory-corruption-probability 0 \
  --diagnostic-validation --validation-interval 4 \
  --log-every 500 --epochs 15 --device cuda \
  --checkpoint-out runs/paired_warmup_10_5_matched/detector.pt

# Joint training with the big model frozen

# Loads the big backbone AND shared LSTM/SSD from the latest stage2.pt.

# Starts joint training next epoch: epoch 7 for the epoch-6 source.
